#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>
#include <unistd.h>

// Function to generate random sensor data
int generateSensorData() {
    return rand() % 61; // Random number between 0 and 60
}

// Define struct for sensor and motor data
struct SensorData {
    // Sensor data
    int e1;
    int e2;
    int e3;
    int e4;
    int e5;
    // Motor data
    int M1;
    int M2;
    int M3;
    int M4;
    int M5;
   };


// Define task periods in microseconds
#define SENSOR_PERIOD_MS 50
#define TRAJECTORY_PERIOD_MS 100
#define MOTOR_PERIOD_MS 10

// Mutexes for thread synchronization
pthread_mutex_t sensorMutex = PTHREAD_MUTEX_INITIALIZER;
pthread_mutex_t trajectoryMutex = PTHREAD_MUTEX_INITIALIZER;
pthread_mutex_t motorMutex = PTHREAD_MUTEX_INITIALIZER;
void print_motor(void*arg){
    // Cast the void pointer to the struct type
    struct SensorData *M = (struct SensorData *)arg;
    printf("Moving to trajectory computed [%i,%i,%i,%i,%i]...\n",M->M1,M->M2,M->M3,M->M4,M->M5);

}

// Sensor data processing task function

    // Cast the void pointer to the struct type
    struct SensorData *M = (struct SensorData *)arg;
    // TODO: Implement the sensor data processing task
    while (1) {

        pthread_mutex_lock(&sensorMutex);

// Trajectory planning task function
void* trajectory_task(void* arg) {
    struct timespec start, end;
    // Cast the void pointer to the struct type
    struct SensorData *M = (struct SensorData *)arg;
    // TODO: Implement the trajectory planning task
    while (1) {
        clock_gettime(CLOCK_MONOTONIC, &start);
        // Acquire trajectory planning mutex
        pthread_mutex_lock(&trajectoryMutex);
        M->M1 = M->e1 - M->e5;
        M->M2 = M->e1 + 2 * M->e2;
        M->M3 = M->e2 + M->e3;
        M->M4 = M->e4 + 4 * M->e3;
        M->M5 = M->e4 - 2 * M->e5;

}

int main() {
    srand(time(NULL));

    struct SensorData M;

    // Threads for each task
    pthread_t motor_thread, sensor_thread, trajectory_thread;
    pthread_attr_t attr;
    
    // Set scheduling policy to SCHED_FIFO for real-time behavior
    struct sched_param param;
    pthread_attr_init(&attr);
    pthread_attr_setschedpolicy(&attr, SCHED_FIFO);

    // Adjust priority as needed

    
    // Create motor control thread
    param.sched_priority = 40; // Example priority
    pthread_attr_setschedparam(&attr, &param);
    pthread_create(&motor_thread, &attr, motor_task,&M);
    
    // Create sensor data processing thread
    param.sched_priority = 30; // Example priority
    pthread_attr_setschedparam(&attr, &param);
    pthread_create(&sensor_thread, &attr, sensor_task,&M);
    
    // Create trajectory planning thread
    param.sched_priority = 20; // Example priority
    pthread_attr_setschedparam(&attr, &param);
    pthread_create(&trajectory_thread, &attr, trajectory_task, &M);
    
    
    // Main thread acts as the real-time task scheduler
    
    // Wait for threads to finish (should never happen) and join them
    pthread_join(sensor_thread, NULL);
    pthread_join(trajectory_thread, NULL);
    pthread_join(motor_thread, NULL);

    
    return 0;
}
